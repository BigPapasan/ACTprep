        function calculateEstimatedScores() {
            // Converts accuracy stats into ACT Scale Scores (1-36 scale)
            const convertToScale = (solved, correct) => {
                if (!solved || solved === 0) return 24; // Default starting estimate
                const pct = correct / solved;
                return Math.min(36, Math.max(12, Math.round(12 + (pct * 24))));
            };

            const eScore = convertToScale(state.stats.English.solved, state.stats.English.correct);
            const mScore = convertToScale(state.stats.Math.solved, state.stats.Math.correct);
            const rScore = convertToScale(state.stats.Reading.solved, state.stats.Reading.correct);
            const sScore = convertToScale(state.stats.Science.solved, state.stats.Science.correct);

            const composite = Math.round((eScore + mScore + rScore + sScore) / 4);

            return { English: eScore, Math: mScore, Reading: rScore, Science: sScore, Composite: composite };
        }

        function updateDashboardUI() {
            const scores = calculateEstimatedScores();

            document.getElementById('dash-composite-score').innerText = scores.Composite;
            document.getElementById('nav-est-score').innerText = scores.Composite;
            document.getElementById('dash-composite-bar').style.width = `${(scores.Composite/36)*100}%`;

            // Questions count total
            let totalSolved = 0;
            let totalCorrect = 0;
            Object.keys(state.stats).forEach(s => {
                totalSolved += state.stats[s].solved;
                totalCorrect += state.stats[s].correct;
            });

            const overallAcc = totalSolved > 0 ? Math.round((totalCorrect / totalSolved) * 100) : 0;
            document.getElementById('dash-questions-count').innerText = totalSolved;
            document.getElementById('dash-accuracy-rate').innerText = `${overallAcc}% Acc`;
            document.getElementById('dash-questions-bar').style.width = `${Math.min(100, totalSolved * 2)}%`;

            // Flashcards
            const masteredCount = Object.values(state.flashcardMastery).filter(Boolean).length;
            document.getElementById('dash-cards-mastered').innerText = masteredCount;
            document.getElementById('dash-cards-total').innerText = `/ ${flashcardsDb.length} Cards`;
            document.getElementById('dash-cards-bar').style.width = `${(masteredCount/flashcardsDb.length)*100}%`;

            // Goal gap
            document.getElementById('dash-target-score').innerText = state.targetScore;
            const gap = state.targetScore - scores.Composite;
            document.getElementById('dash-score-gap').innerText = gap <= 0 ? "Goal Reached! 🎉" : `+${gap} pts needed`;

            // Subject level UI
            ['English', 'Math', 'Reading', 'Science'].forEach(sub => {
                const subKey = sub.toLowerCase();
                const score = scores[sub];
                const solved = state.stats[sub].solved;
                const correct = state.stats[sub].correct;
                const acc = solved > 0 ? Math.round((correct/solved)*100) : 0;

                const scoreEl = document.getElementById(`dash-score-${subKey}`);

                if (accEl) accEl.innerText = `${acc}%`;
            });

            // Radar Chart Render
            renderRadarChart(scores);

        function selectAnswer(optionIdx) {
            if (selectedOptionIdx !== null) return; // Prevent changing after selected
            selectedOptionIdx = optionIdx;

            const q = currentFilteredQuestions[currentQuestionIdx];
            const isCorrect = (optionIdx === q.correct);

            // Update stats
            state.stats[q.subject].solved += 1;
            if (isCorrect) state.stats[q.subject].correct += 1;
            saveState();

            // Highlight Options UI
            const container = document.getElementById('q-options-container');
            const buttons = container.children;

            for (let i = 0; i < buttons.length; i++) {
                buttons[i].disabled = true;
                if (i === q.correct) {
                    buttons[i].className = "w-full text-left p-4 rounded-xl border border-emerald-500/80 bg-emerald-950/40 text-emerald-200 text-sm font-medium flex items-center space-x-3";
                } else if (i === optionIdx && !isCorrect) {
                    buttons[i].className = "w-full text-left p-4 rounded-xl border border-rose-500/80 bg-rose-950/40 text-rose-200 text-sm font-medium flex items-center space-x-3";
                }
            }

            // Show Explanation Drawer
            const expBox = document.getElementById('q-explanation-box');
            const expBadge = document.getElementById('q-explanation-badge');
            const expText = document.getElementById('q-explanation-text');

            expBox.classList.remove('hidden');
            if (isCorrect) {
                expBadge.innerText = "Correct! 🎉";
                expBadge.className = "px-2.5 py-0.5 rounded text-xs font-bold bg-emerald-500/20 text-emerald-400 border border-emerald-500/30";
                if (typeof confetti === 'function') {
                    confetti({ particleCount: 40, spread: 60, origin: { y: 0.8 } });
                }
            } else {
                expBadge.innerText = "Incorrect";
                expBadge.className = "px-2.5 py-0.5 rounded text-xs font-bold bg-rose-500/20 text-rose-400 border border-rose-500/30";
            }
            expText.innerText = q.explanation;
        }

        // Practice Timer Logic
        function startTimer() {
            if (timerInterval) clearInterval(timerInterval);
            timerInterval = setInterval(() => {
                timerSeconds++;
                const m = String(Math.floor(timerSeconds / 60)).padStart(2, '0');
                const s = String(timerSeconds % 60).padStart(2, '0');
                document.getElementById('practice-timer').innerText = `${m}:${s}`;
            }, 1000);
        }

        function toggleTimer() {
            const btn = document.getElementById('btn-timer-toggle');
            if (timerInterval) {
                clearInterval(timerInterval);
                timerInterval = null;
                btn.innerText = "Resume Timer";
            } else {
                startTimer();
                btn.innerText = "Pause Timer";
            }
        }

        /* CANVAS SCRATCHPAD LOGIC                              */

        let canvas, ctx;
        let isDrawing = false;
        let strokeColor = '#38bdf8';

        function initScratchpad() {
            canvas = document.getElementById('scratchpadCanvas');
            if (!canvas) return;
            ctx = canvas.getContext('2d');

            // Responsive size check
            canvas.width = canvas.parentElement.clientWidth;
            canvas.height = 320;

            ctx.lineWidth = 3;
            ctx.lineCap = 'round';
            ctx.strokeStyle = strokeColor;

            // Mouse Events
            canvas.onmousedown = (e) => { isDrawing = true; ctx.beginPath(); ctx.moveTo(e.offsetX, e.offsetY); };
            canvas.onmousemove = (e) => { if (isDrawing) { ctx.lineTo(e.offsetX, e.offsetY); ctx.stroke(); } };
            canvas.onmouseup = () => { isDrawing = false; };

            // Touch Events for Mobile
            canvas.ontouchstart = (e) => {
                isDrawing = true;
                const rect = canvas.getBoundingClientRect();
                ctx.beginPath();
                ctx.moveTo(e.touches[0].clientX - rect.left, e.touches[0].clientY - rect.top);
            };
            canvas.ontouchmove = (e) => {
                if (isDrawing) {
                    const rect = canvas.getBoundingClientRect();
                    ctx.lineTo(e.touches[0].clientX - rect.left, e.touches[0].clientY - rect.top);
                    ctx.stroke();
                }
            };
            canvas.ontouchend = () => { isDrawing = false; };
        }

        function setCanvasColor(color) {
            strokeColor = color;
            if (ctx) ctx.strokeStyle = strokeColor;
        }

        function clearCanvas() {
            if (ctx && canvas) {
                ctx.clearRect(0, 0, canvas.width, canvas.height);
            }
        }

        let calcExpr = "";

        function calcNum(n) {
            calcExpr += n;
            updateCalcDisplay();
        }

        function calcOp(op) {
            calcExpr += op;
            updateCalcDisplay();
        }

        function calcFunc(fn) {
            if (fn === 'sqrt') calcExpr += "Math.sqrt(";
            if (fn === 'sin') calcExpr += "Math.sin(";
            if (fn === 'cos') calcExpr += "Math.cos(";
            if (fn === 'tan') calcExpr += "Math.tan(";
            updateCalcDisplay();
        }

        function calcClear() {
            calcExpr = "";
            document.getElementById('calc-display').innerText = "0";
            document.getElementById('calc-expression').innerText = "";
        }

        function calcEquals() {
            try {
                const res = eval(calcExpr);
                document.getElementById('calc-expression').innerText = calcExpr;
                document.getElementById('calc-display').innerText = Number.isInteger(res) ? res : res.toFixed(4);
                calcExpr = String(res);
            } catch(e) {
                document.getElementById('calc-display').innerText = "Error";
                calcExpr = "";
            }
        }

        function updateCalcDisplay() {
            document.getElementById('calc-display').innerText = calcExpr || "0";
        }
